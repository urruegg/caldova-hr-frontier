# ----------------------------------------------------------------------------- SVG

def glyph_body(glyphs: dict, name: str, style: str, colour: str) -> str:
    key = f"{name}-24-{style}"
    if key not in glyphs["icons"]:
        raise KeyError(f"Glyph {key} is not in scripts/glyphs/fluent-system-icons-subset.json")
    return glyphs["icons"][key]["body"].replace("currentColor", colour)


def colour_svg(agent: dict, palette: dict, glyphs: dict, env: str, size: int = 192) -> str:
    dom = palette["domains"][agent["journey"]]
    body = glyph_body(glyphs, agent["glyph"], "filled", dom["glyph"])
    parts = [f'<svg xmlns="http://www.w3.org/2000/svg" width="{size}" height="{size}" viewBox="0 0 192 192">',
             f'<rect width="192" height="192" fill="{dom["background"]}"/>',
             f'<g transform="translate(48 48) scale(4)">{body}</g>']
    e = palette["environments"].get(env)
    if e and agent.get("environmentScoped", True):
        parts += [f'<rect x="0" y="152" width="192" height="4" fill="{e["separator"]}"/>',
                  f'<rect x="0" y="156" width="192" height="36" fill="{e["band"]}"/>',
                  f'<text x="96" y="182.5" text-anchor="middle" font-family="{html.escape(FONT)}" '
                  f'font-weight="700" font-size="22" letter-spacing="3" fill="{e["label"]}">{e["text"]}</text>']
    parts.append("</svg>")
    return "".join(parts)


def outline_svg(agent: dict, glyphs: dict, size: int = 32) -> str:
    body = glyph_body(glyphs, agent["glyph"], "regular", "#FFFFFF")
    return (f'<svg xmlns="http://www.w3.org/2000/svg" width="{size}" height="{size}" viewBox="-2 -2 28 28">'
            f'{body}</svg>')


# ----------------------------------------------------------------------------- helpers

def png_info(path: Path):
    with open(path, "rb") as f:
        head = f.read(24)
    if head[:8] != b"\x89PNG\r\n\x1a\n":
        raise ValueError(f"{path} is not a PNG")
    w, h = struct.unpack(">II", head[16:24])
    return w, h, path.stat().st_size


def data_uri_png(path: Path) -> str:
    return "data:image/png;base64," + base64.b64encode(path.read_bytes()).decode("ascii")


def esc(t) -> str:
    return html.escape(str(t if t is not None else ""))


def sized(svg: str, px: int) -> str:
    """Display an inline 192-unit icon SVG at another pixel size (viewBox keeps the geometry)."""
    return svg.replace('width="192" height="192"', f'width="{px}" height="{px}"', 1)


def chip(status_class: str, label: str | None = None, size: int = 16) -> str:
    bg, fg, default = STATUS_STYLE.get(status_class, STATUS_STYLE["hypothesis"])
    return (f'<span style="display:inline-block;padding:4px 12px;border-radius:999px;background:{bg};color:{fg};'
            f'font-size:{size}px;font-weight:700;white-space:nowrap">{esc(label or default)}</span>')


def short_status(agent: dict) -> str:
    if agent["statusClass"] == "mvp" and agent.get("mvp"):
        return agent["mvp"]
    return STATUS_STYLE.get(agent["statusClass"], STATUS_STYLE["hypothesis"])[2]

def identity_record(cfg: dict, palette: dict, agent: dict) -> dict:
    d = agent["derived"]
    envs = L.ENVS if agent.get("environmentScoped", True) else ("PROD",)
    files = {}
    for env in envs:
        sfx = "" if env == "PROD" else f"-{env.lower()}"
        files[env] = {"color192": f"color-192{sfx}.png", "color32": f"color-32{sfx}.png",
                      "outline32": "outline-32.png", "svg": f"icon{sfx}.svg"}
    rec = {
        "id": agent["id"],
        "kind": agent["kind"],
        "role": agent["role"],
        "journey": {"id": agent["journey"], "name": agent["journeyName"]},
        "tokens": {"po": agent["po"], "domain": agent["domain"], "scope": agent.get("scope", "")},
        "displayName": {env: d["displayName"][env] for env in envs},
        "shortDescription": {env: L.env_description(cfg, agent, env, "short") for env in envs},
        "fullDescription": {env: L.env_description(cfg, agent, env, "full") for env in envs},
        "slug": d["slug"],
        "schemaName": d["schemaName"],
        "solution": None if not d["solutionUniqueName"] else
        {"uniqueName": d["solutionUniqueName"], "displayName": d["solutionDisplayName"]},
        "envVarPrefix": d["envVarPrefix"],
        "connectionContext": d["connectionContext"],
        "flowNamePrefix": d["flowNamePrefix"],
        "github": None if not d.get("githubAgentName") else
        {"agentName": d["githubAgentName"], "file": d["githubAgentFile"], "proposedName": bool(agent.get("githubAgentProposed"))},
        "icon": {"background": d["icon"]["background"], "glyphColour": d["icon"]["glyphColor"],
                 "contrast": d["icon"]["contrast"], "glyph": f'Fluent UI System Icons: {agent["glyph"]} (filled / regular)',
                 "files": files},
        "teams": None if not d["teams"] else {
            "shortName": {env: d["teams"]["shortName"][env] for env in envs},
            "fullName": {env: d["teams"]["fullName"][env] for env in envs},
            "shortDescription": {env: L.env_description(cfg, agent, env, "short") for env in envs},
            "fullDescription": {env: L.env_description(cfg, agent, env, "full") for env in envs},
            "accentColor": d["teams"]["accentColor"],
            "developer": {"name": "Georg Fischer Ltd", "websiteUrl": "https://www.georgfischer.com",
                          "privacyUrl": "[GF to provide]", "termsOfUseUrl": "[GF to provide]"},
        },
        "status": agent["status"],
        "statusClass": agent["statusClass"],
        "audience": agent["audience"],
        "processOwnerArea": L.po_area(cfg, agent["po"])["processOwner"],
        "owner": agent["owner"],
        "channels": agent["channels"],
        "gfName": agent.get("gfName"),
        "blueprintName": agent["blueprintName"],
        "evidence": agent["evidence"],
        "notes": agent.get("notes"),
        "generatedFrom": f'config/agent-identity.json v{cfg["version"]}',
    }
    return rec

def page(body: str, width: int, height: int, bg: str = "#FFFFFF") -> str:

            f'html,body{{margin:0;padding:0}}body{{width:{width}px;height:{height}px;overflow:hidden;background:{bg};'
            f'font-family:{FONT};color:#282828;position:relative}}'
            f'.mono{{font-family:{MONO}}}</style></head><body>{body}</body></html>')

    return page(body, 1880, 1010)


# ----------------------------------------------------------------------------- catalog markdown

def catalog_md(cfg, palette, agents_by_j) -> str:
    out = []
    out.append("# Agent identity catalog\n")
    out.append(f"Generated from `config/agent-identity.json` v{cfg['version']} ({cfg['date']}) by `scripts/build_identity_assets.py`. "
               "Do not edit by hand: change the tokens in the config and rebuild.\n")
    out.append("Display name = `<PO> <Domain> [Scope] [DEV | TEST]`; PROD has no postfix. "
               "Technical names never carry an environment. Status values come from the blueprint v0.4.\n")
    total = sum(len(v) for v in agents_by_j.values())
    mvp = [a for v in agents_by_j.values() for a in v if a["statusClass"] == "mvp"]
    out.append(f"**{total} agents** in 6 journeys · **{len(mvp)} MVP agents** for the prototype days · "
               f"publisher prefix `{cfg['naming']['publisherPrefix']}` (decision G5 open).\n")
    for jid, items in agents_by_j.items():
        dom = palette["domains"][jid]
        out.append(f"\n## {jid} · {dom['name']}\n")
        out.append(f"Colour **{dom['brandColour']}** `{dom['background']}` · glyph `{dom['glyph']}` · contrast {dom['contrast']}:1 · PO area(s): {dom['poAreas']}\n")
        out.append("| Icon | PROD name | DEV | TEST | PO · Domain | Status | Schema / GitHub agent | Solution | Folder |")
        out.append("|---|---|---|---|---|---|---|---|---|")
        for a in items:
            d = a["derived"]
            icon = f'<img src="../agents/{d["slug"]}/identity/color-192.png" width="40" alt="{esc(d["displayName"]["PROD"])} icon">'
            scoped = a.get("environmentScoped", True)
            dev = f'`{d["displayName"]["DEV"]}`' if scoped else "n/a"
            test = f'`{d["displayName"]["TEST"]}`' if scoped else "n/a"
            tech = f'`{d["schemaName"]}`' if d["schemaName"] else f'`{d["githubAgentFile"]}`'
            sol = f'`{d["solutionUniqueName"]}`' if d["solutionUniqueName"] else "n/a (repository)"
            potok = f'{a["po"]} · {a["domain"]}' + (f' · {a["scope"]}' if a.get("scope") else "")
            out.append(f'| {icon} | **{d["displayName"]["PROD"]}** | {dev} | {test} | {potok} | {short_status(a)} | {tech} | {sol} | `agents/{d["slug"]}/` |')
        for a in items:
            d = a["derived"]
            out.append(f'\n### {d["displayName"]["PROD"]} ({a["id"]})\n')
            out.append(f'<img src="../agents/{d["slug"]}/identity/card.png" width="600" alt="Identity card for {esc(d["displayName"]["PROD"])}">\n')
            out.append(f'- **Purpose (short, ≤ 80 with env marker):** {a["shortDescription"]}')
            out.append(f'- **Full description:** {a["fullDescription"]}')
            out.append(f'- **Audience:** {a["audience"]}')
            out.append(f'- **Process owner area:** {L.po_area(cfg, a["po"])["processOwner"]}')
            out.append(f'- **Owner (blueprint):** {a["owner"] or "to confirm"}')
            if a.get("gfName"):
                out.append(f'- **GF working name:** {a["gfName"]}')
            out.append(f'- **Channels:** {", ".join(a["channels"]) or "to confirm"}')
            out.append(f'- **Status:** {a["status"]}')
            if d["schemaName"]:
                out.append(f'- **Technical:** schema `{d["schemaName"]}` · solution `{d["solutionUniqueName"]}` ("{d["solutionDisplayName"]}") · '
                           f'variables `{d["envVarPrefix"]}…` · connection references `{cfg["naming"]["publisherPrefix"]}_<Connector>{d["connectionContext"]}` · flows "{d["flowNamePrefix"]}<Action>"')
            if d.get("githubAgentName"):
                prop = " (name proposed, not in the factory concept)" if a.get("githubAgentProposed") else ""
                out.append(f'- **GitHub Copilot custom agent:** `{d["githubAgentFile"]}`{prop}')
            out.append(f'- **Icon:** glyph `{a["glyph"]}` on `{d["icon"]["background"]}` · files in `agents/{d["slug"]}/identity/`')
            if a.get("notes"):
                out.append(f'- **Note:** {a["notes"]}')
            ev = ", ".join(a["evidence"]) if a["evidence"] else "blueprint v0.4 (hypothesis)"
            out.append(f'- **Evidence:** {ev}')
    out.append("")
    return "\n".join(out)


# ----------------------------------------------------------------------------- main

def main() -> int:
    ap = argparse.ArgumentParser(description=__doc__, formatter_class=argparse.RawDescriptionHelpFormatter)
    ap.add_argument("--root", default=str(L.ROOT))
    ap.add_argument("--no-render", action="store_true", help="skip PNG rendering (no Node.js needed)")
    args = ap.parse_args()
    root = Path(args.root).resolve()
    cfg, palette = L.load_config(root)
    glyphs = L.load_json(root / "scripts" / "glyphs" / "fluent-system-icons-subset.json")
    logo_uri = data_uri_png(root / "brand" / "gf-logo-blue.png")

    # 1. derive names (token edits propagate everywhere)
    for a in cfg["agents"]:
        a["derived"] = L.derive(cfg, a, palette)
    L.save_json(root / "config" / "agent-identity.json", cfg)

    build = Path(tempfile.mkdtemp(prefix="gf-identity-"))
    (build / "html").mkdir(parents=True)
    jobs = []
    svgs_all = {}
    agents_by_j = {}
    for a in cfg["agents"]:
        agents_by_j.setdefault(a["journey"], []).append(a)
        d = a["derived"]
        folder = root / "agents" / d["slug"] / "identity"
        folder.mkdir(parents=True, exist_ok=True)
        for old in list(folder.glob("*.png")) + list(folder.glob("*.svg")) + list(folder.glob("identity.json")):
            old.unlink()  # regenerate cleanly; the folder itself is kept
        envs = L.ENVS if a.get("environmentScoped", True) else ("PROD",)
        svgs = {}
        for env in envs:
            sfx = "" if env == "PROD" else f"-{env.lower()}"
            svg = colour_svg(a, palette, glyphs, env)
            svgs[env] = svg
            (folder / f"icon{sfx}.svg").write_text(svg, encoding="utf-8")
            small = build / "html" / f"{d['slug']}{sfx}-32.svg"
            small.write_text(colour_svg(a, palette, glyphs, env, size=32), encoding="utf-8")
            jobs.append({"src": str(folder / f"icon{sfx}.svg"), "out": str(folder / f"color-192{sfx}.png"), "width": 192, "height": 192, "transparent": False})
            jobs.append({"src": str(small), "out": str(folder / f"color-32{sfx}.png"), "width": 32, "height": 32, "transparent": False})
        out_svg = outline_svg(a, glyphs)
        svgs["outline"] = out_svg
        (folder / "outline.svg").write_text(out_svg, encoding="utf-8")
        jobs.append({"src": str(folder / "outline.svg"), "out": str(folder / "outline-32.png"), "width": 32, "height": 32, "transparent": True})
        svgs_all[a["id"]] = svgs
        L.save_json(folder / "identity.json", identity_record(cfg, palette, a))
        card = build / "html" / f"card-{d['slug']}.html"
        card.write_text(card_html(cfg, palette, a, svgs, logo_uri), encoding="utf-8")
        jobs.append({"src": str(card), "out": str(folder / "card.png"), "width": 1200, "height": 630, "transparent": False})

    vis = root / "visuals"
    vis.mkdir(exist_ok=True)
    ov = build / "html" / "overview.html"
    ov.write_text(overview_html(cfg, palette, agents_by_j, svgs_all, logo_uri), encoding="utf-8")
    jobs.append({"src": str(ov), "out": str(vis / "gf-hr-agent-family.png"), "width": 2400, "height": 1420, "transparent": False})
    mvps = sorted([a for a in cfg["agents"] if a["statusClass"] == "mvp"], key=lambda a: a.get("mvp") or "")
    ms = build / "html" / "mvp.html"
    ms.write_text(mvp_sheet_html(cfg, palette, mvps, svgs_all, logo_uri), encoding="utf-8")
    jobs.append({"src": str(ms), "out": str(vis / "gf-hr-mvp-identity-sheet.png"), "width": 2100, "height": 1080, "transparent": False})
    sample = {}
    for jid, items in agents_by_j.items():
        lead = next((x for x in items if x["role"] == "orchestrator"), items[0])
        sample[jid] = svgs_all[lead["id"]]["PROD"]
    pv = build / "html" / "palette.html"
    pv.write_text(palette_html(cfg, palette, sample, logo_uri), encoding="utf-8")
    jobs.append({"src": str(pv), "out": str(vis / "gf-hr-colour-schema.png"), "width": 1880, "height": 1010, "transparent": False})

    (root / "docs").mkdir(exist_ok=True)
    (root / "docs" / "agent-identity-catalog.md").write_text(catalog_md(cfg, palette, agents_by_j), encoding="utf-8")

    if not args.no_render:
        jf = build / "jobs.json"
        jf.write_text(json.dumps(jobs, indent=1), encoding="utf-8")
        r = subprocess.run(["node", str(root / "scripts" / "render_png.js"), str(jf)], capture_output=True, text=True)
        print(r.stdout.strip())
        if r.returncode != 0:
            print(r.stderr, file=sys.stderr)
            return r.returncode
        problems = []
        for j in jobs:
            w, h, size = png_info(Path(j["out"]))
            if (w, h) != (j["width"], j["height"]):
                problems.append(f'{j["out"]}: {w}x{h}')
            if j["width"] <= 192 and size > 100 * 1024:
                problems.append(f'{j["out"]}: {size} bytes > 100 KB')
        if problems:
            print("\n".join(problems), file=sys.stderr)
            return 1
    shutil.rmtree(build, ignore_errors=True)
    print(f"identity assets built for {len(cfg['agents'])} agents")
    return 0


if __name__ == "__main__":
    sys.exit(main())
