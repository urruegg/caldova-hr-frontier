// Renders SVG and HTML files to PNG with a headless Chromium (Playwright).
// Usage: node scripts/render_png.js <jobs.json>
//   jobs.json: [{ "src": "<absolute path to .svg or .html>", "out": "<png path>", "width": 192, "height": 192, "transparent": false }]
// Environment (optional):
//   PLAYWRIGHT_MODULE  path to the playwright package if it is not resolvable with require('playwright')
//   CHROMIUM_PATH      path to a Chromium / headless shell executable (otherwise Playwright's own browser)
const fs = require('fs');
const path = require('path');
const { pathToFileURL } = require('url');

function loadPlaywright() {
  const candidates = [process.env.PLAYWRIGHT_MODULE, 'playwright', 'playwright-core'].filter(Boolean);
  for (const c of candidates) {
    try { return require(c); } catch (e) { /* try next */ }
  }
  throw new Error('Playwright not found. Run "npm i -D playwright && npx playwright install chromium" or set PLAYWRIGHT_MODULE.');
}

(async () => {
  const jobsFile = process.argv[2];
  if (!jobsFile) { console.error('usage: node render_png.js <jobs.json>'); process.exit(2); }
  const jobs = JSON.parse(fs.readFileSync(jobsFile, 'utf8'));
  const { chromium } = loadPlaywright();
  const launch = process.env.CHROMIUM_PATH ? { executablePath: process.env.CHROMIUM_PATH } : {};
  const browser = await chromium.launch(launch);
  const page = await browser.newPage({ deviceScaleFactor: 1 });
  let n = 0;
  for (const job of jobs) {
    await page.setViewportSize({ width: job.width, height: job.height });
    await page.goto(pathToFileURL(path.resolve(job.src)).href, { waitUntil: 'load' });
    fs.mkdirSync(path.dirname(job.out), { recursive: true });
    await page.screenshot({ path: job.out, omitBackground: !!job.transparent,
                            clip: { x: 0, y: 0, width: job.width, height: job.height } });
    n += 1;
  }
  await browser.close();
  console.log(`rendered ${n} image(s)`);
})().catch(err => { console.error(err); process.exit(1); });
